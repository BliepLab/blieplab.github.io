Een kleine, respectvolle voorziening binnen jullie buurt-mesh: geen camera, geen logboek, alleen één dagelijkse controle en een afgesproken zorgkring van 2–3 mensen.A small, respectful provision inside your neighbourhood mesh: no camera, no log, just one daily check and an agreed care circle of 2–3 people.Небольшая и уважительная функция внутри вашей соседской mesh-сети: без камеры и журнала, только одна ежедневная проверка и согласованный круг заботы из 2–3 человек.Невелика й шаноблива функція всередині вашої сусідської mesh-мережі: без камери та журналу, лише одна щоденна перевірка й узгоджене коло турботи з 2–3 людей.Mała, pełna szacunku funkcja w waszej sąsiedzkiej sieci mesh: bez kamery i dziennika, tylko jedna codzienna kontrola i uzgodniony krąg opieki z 2–3 osób.

Belangrijk: dit is een sociale check, geen medisch hulpmiddel. Het ziet geen val, geen hartstilstand en geen nood zolang iemand wel actief is. Wie medische risico’s heeft, houdt een personenalarmering (alarmknop met meldkamer). Bij nood: bel 112.Important: this is a social check, not a medical device. It does not see a fall, a cardiac arrest or distress as long as someone is active. Anyone with medical risks keeps a personal alarm service (alarm button with a control room). In an emergency: call 112.Важно: это социальная проверка, а не медицинское устройство. Оно не видит падение, остановку сердца или беду, пока человек активен. Тем, у кого есть медицинские риски, нужна служба персональной тревожной кнопки (с диспетчерским центром). В экстренной ситуации: звоните 112.Важливо: це соціальна перевірка, а не медичний пристрій. Він не бачить падіння, зупинку серця чи біду, поки людина активна. Тим, хто має медичні ризики, потрібна служба персональної тривожної кнопки (з диспетчерським центром). У надзвичайній ситуації: телефонуйте 112.Ważne: to kontrola społeczna, a nie wyrób medyczny. Nie widzi upadku, zatrzymania krążenia ani nieszczęścia, dopóki ktoś jest aktywny. Osoby z ryzykiem medycznym potrzebują usługi osobistego przycisku alarmowego (z centrum dyspozytorskim). W nagłej sytuacji: dzwoń pod 112.

Достаточно нажать кнопку или просто войти в комнату.Якщо активності не було, пристрій спочатку тихо пищить біля самого мешканця.   • Pas als er na 30 minuten nog niets is, krijgt de zorgkring één kort bericht. A person calls or walks over.Только если через 30 минут всё ещё ничего нет, круг заботы получает одно короткое сообщение. Человек звонит или заходит.Лише якщо за 30 хвилин усе ще нічого немає, коло турботи отримує одне коротке повідомлення. // Reads a presence sensor, decides with checkin_core.h and sends a short text
// to a Meshtastic node over a serial cable (Meshtastic "Serial module", TEXTMSG mode).
//
// Libraries: "RTClib" by Adafruit. Board: ESP32C3 Dev Module (USB CDC on boot: enabled).
// Put checkin_core.h in the same folder as this file.
//
// IMPORTANT: this sketch was written for real hardware but has not been tested on it yet.
// Test every part with the checklist on the guide page before you rely on it.

#include <Wire.h>
#include <RTClib.h>
#include "checkin_core.h"

// ---------- settings you change ----------
#define HOUSE "12"                  // house number used in the messages
#define SERIAL_BAUD 9600            // must match the Meshtastic Serial module setting
checkin::Config cfg;                // times: see checkin_core.h (07:00, 10:30, 30 min)

// ---------- pins (ESP32-C3 SuperMini) ----------
const int PIN_PRESENCE = 0;   // LD2410C OUT (HIGH = presence)
const int PIN_BUTTON   = 1;   // big button to GND
const int PIN_AWAY     = 3;   // "I'm away" switch to GND (closed = away)
const int PIN_BUZZER   = 4;   // active buzzer (use a transistor if it draws more than 20 mA)
const int PIN_LED      = 5;   // LED + 330 ohm to GND
const int PIN_SDA      = 6;   // DS3231
const int PIN_SCL      = 7;
const int PIN_MESH_TX  = 10;  // to the RX pin of the Meshtastic node (common GND!)

// ---------- messages (Dutch; change to your language) ----------
const char MSG_OK[]        = "Check-in OK nr " HOUSE;
const char MSG_ALERT[]     = "ALARM check-in gemist nr " HOUSE ": nog geen activiteit vandaag. Bel of loop even langs.";
const char MSG_REPEAT[]    = "HERHALING check-in gemist nr " HOUSE ". Bel of loop even langs.";
const char MSG_RECOVERED[] = "Check-in nr " HOUSE ": weer activiteit, alles in orde.";
const char MSG_TEST[]      = "TEST check-in nr " HOUSE;

RTC_DS3231 rtc;
checkin::State state;
uint32_t lastTick = 0, lastBeep = 0, buttonDownAt = 0;
uint8_t presenceStreak = 0;
bool buttonWas = false;
bool buttonEvent = false;

void sendText(const char* text) {
  Serial1.println(text);
  Serial.println(text);  // also visible on the USB serial monitor
}

void beep(uint16_t ms) {
  digitalWrite(PIN_BUZZER, HIGH);
  delay(ms);
  digitalWrite(PIN_BUZZER, LOW);
}

void setup() {
  pinMode(PIN_PRESENCE, INPUT);
  pinMode(PIN_BUTTON, INPUT_PULLUP);
  pinMode(PIN_AWAY, INPUT_PULLUP);
  pinMode(PIN_BUZZER, OUTPUT);
  pinMode(PIN_LED, OUTPUT);
  Serial.begin(115200);
  Serial1.begin(SERIAL_BAUD, SERIAL_8N1, -1, PIN_MESH_TX);  // transmit only
  Wire.begin(PIN_SDA, PIN_SCL);
  if (!rtc.begin()) {
    while (true) { digitalWrite(PIN_LED, !digitalRead(PIN_LED)); delay(100); }  // fast blink: no RTC
  }
  if (rtc.lostPower()) {
    // First start: take the time of compiling. Flash the board right after compiling,
    // or set the time another way, otherwise the clock is wrong.
    rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
  }
  beep(100);
}

void loop() {
  // button: short press = "I'm OK", long press (3 s) = send a test message
  bool down = digitalRead(PIN_BUTTON) == LOW;
  if (down && !buttonWas) buttonDownAt = millis();
  if (!down && buttonWas) {
    if (millis() - buttonDownAt >= 3000) { sendText(MSG_TEST); beep(300); }
    else buttonEvent = true;
  }
  buttonWas = down;

  // once per second: sensor, clock, decision
  if (millis() - lastTick >= 1000) {
    lastTick = millis();
    presenceStreak = (digitalRead(PIN_PRESENCE) == HIGH) ? min<int>(presenceStreak + 1, 10) : 0;
    DateTime now = rtc.now();
    checkin::Input in;
    in.minuteOfDay = now.hour() * 60 + now.minute();
    in.activity = presenceStreak >= 2 || buttonEvent;  // 2 s of presence filters out glitches
    in.away = digitalRead(PIN_AWAY) == LOW;
    buttonEvent = false;

    switch (checkin::step(state, cfg, in)) {
      case checkin::EV_SEND_OK:           sendText(MSG_OK); break;
      case checkin::EV_SEND_ALERT:        sendText(MSG_ALERT); break;
      case checkin::EV_SEND_ALERT_REPEAT: sendText(MSG_REPEAT); break;
      case checkin::EV_SEND_RECOVERED:    sendText(MSG_RECOVERED); break;
      default: break;
    }

    // status LED: steady = fine, slow blink = prompting, fast blink = alert sent, off = away
    uint8_t sec = now.second();
    switch (state.phase) {
      case checkin::PROMPTING: digitalWrite(PIN_LED, sec % 2); break;
      case checkin::ALERTED:   digitalWrite(PIN_LED, (millis() / 250) % 2); break;
      case checkin::PAUSED:    digitalWrite(PIN_LED, LOW); break;
      default:                 digitalWrite(PIN_LED, sec % 10 == 0 ? HIGH : LOW); break;  // short flash every 10 s
    }
    // gentle beep once a minute while prompting
    if (state.phase == checkin::PROMPTING && sec == 0) beep(150);
  }
}

// checkin_core.h - pure decision logic for the neighbour check-in node.
// No Arduino dependencies, so it can be tested on a normal computer.
// Time is handled in minutes since midnight (0..1439), local time.
#pragma once
#include <stdint.h>

namespace checkin {

struct Config {
  uint16_t windowStartMin = 7 * 60;       // activity counts from 07:00
  uint16_t deadlineMin    = 10 * 60 + 30; // first check at 10:30
  uint16_t graceMin       = 30;           // beeping time before an alert is sent
  uint8_t  alertRepeats   = 2;            // extra alert messages after the first
  uint16_t repeatEveryMin = 30;           // minutes between alert messages
  bool     dailyOk        = true;         // send a short "OK" every day
};

enum Phase : uint8_t {
  WAITING,    // window open or not yet started, waiting for the deadline
  OK_DONE,    // today's check passed (or was skipped)
  PROMPTING,  // no activity at the deadline: beep and wait for the resident
  ALERTED,    // grace time over: care circle has been messaged
  PAUSED      // "I'm away" switch is on
};

enum Event : uint8_t {
  EV_NONE,
  EV_SEND_OK,
  EV_SEND_ALERT,
  EV_SEND_ALERT_REPEAT,
  EV_SEND_RECOVERED
};

struct Input {
  uint16_t minuteOfDay;  // 0..1439
  bool activity;         // presence detected or button pressed right now
  bool away;             // "I'm away" switch is on
};

struct State {
  bool     init = false;
  Phase    phase = WAITING;
  bool     activity = false;   // activity seen inside today's window
  bool     awayPrev = false;
  uint16_t prev = 0;           // previous minuteOfDay
  uint16_t promptStart = 0;
  uint16_t lastAlert = 0;
  uint8_t  alertsSent = 0;
};

// minutes elapsed from 'from' to 'to', across midnight
inline uint16_t elapsed(uint16_t from, uint16_t to) {
  return (uint16_t)((to + 1440u - from) % 1440u);
}

// true when the clock moved over 'mark' between 'prev' and 'now'
inline bool crossed(uint16_t prev, uint16_t now, uint16_t mark) {
  if (prev == now) return false;
  if (prev < now) return prev < mark && now >= mark;
  return mark > prev || mark <= now;  // wrapped over midnight
}

inline bool inWindow(const Config& c, uint16_t now) {
  return now >= c.windowStartMin && now < c.deadlineMin;
}

// Call this every second or so. Returns an event when a message must be sent.
inline Event step(State& s, const Config& c, const Input& in) {
  Event ev = EV_NONE;
  const uint16_t now = in.minuteOfDay;

  if (!s.init) {
    s.init = true;
    s.prev = now;
    s.awayPrev = in.away;
    s.activity = false;
    s.phase = in.away ? PAUSED : (now >= c.deadlineMin ? OK_DONE : WAITING);
  }

  // "I'm away" switch
  if (in.away) {
    s.phase = PAUSED;
    s.awayPrev = true;
    s.prev = now;
    return EV_NONE;
  }
  if (s.awayPrev) {  // switch just turned off: resident is back
    s.awayPrev = false;
    s.activity = true;
    s.alertsSent = 0;
    s.phase = (now >= c.deadlineMin) ? OK_DONE : WAITING;
  }

  // new day: reset when the window start is passed
  if (crossed(s.prev, now, c.windowStartMin)) {
    s.phase = WAITING;
    s.activity = false;
    s.alertsSent = 0;
  }
  s.prev = now;

  // activity
  if (in.activity) {
    if (s.phase == WAITING && inWindow(c, now)) {
      s.activity = true;
    } else if (s.phase == PROMPTING) {
      s.activity = true;
      s.phase = OK_DONE;
    } else if (s.phase == ALERTED) {
      s.activity = true;
      s.phase = OK_DONE;
      s.alertsSent = 0;
      return EV_SEND_RECOVERED;
    }
  }

  // deadline reached
  if (s.phase == WAITING && now >= c.deadlineMin) {
    if (s.activity) {
      s.phase = OK_DONE;
      if (c.dailyOk) ev = EV_SEND_OK;
    } else {
      s.phase = PROMPTING;
      s.promptStart = now;
    }
    return ev;
  }

  // beeping time over
  if (s.phase == PROMPTING && elapsed(s.promptStart, now) >= c.graceMin) {
    s.phase = ALERTED;
    s.alertsSent = 1;
    s.lastAlert = now;
    return EV_SEND_ALERT;
  }

  // repeat alert a limited number of times
  if (s.phase == ALERTED && s.alertsSent <= c.alertRepeats &&
      elapsed(s.lastAlert, now) >= c.repeatEveryMin) {
    s.alertsSent++;
    s.lastAlert = now;
    return EV_SEND_ALERT_REPEAT;
  }
  return ev;
}

}  // namespace checkin
